// ============================================================
//  本地私有配置模板
//  用法：把本文件复制成同一目录下的 LocalSecrets.cs（去掉 .txt 后缀），
//        再把下面的地址换成你自己的服务器。
//
//  为什么单独拆一个文件：
//    公告、更新检查、账号、联机都指向同一台服务器，仓库里不该出现真实地址，
//    所以把它挪到 LocalSecrets.cs —— 那个文件在 .gitignore 里，不会被提交。
// ============================================================

namespace LinxLauncher.Services;

internal static class LocalSecrets
{
    /// <summary>官方服务器地址：公告、更新检查、账号、联机都走这一台。</summary>
    public const string ServerUrl = "https://your-server.example.com";
}
